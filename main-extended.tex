\documentclass[10pt]{article}

% arxiv.sty already loads geometry AND re-applies its own layout via an
% \AtBeginDocument{\newgeometry{textheight=9in, textwidth=6.5in, top=1in,
% headheight=14pt, headsep=25pt, footskip=30pt}}. To shave a small amount
% of bottom margin we override later (\AtBeginDocument fires in
% registration order) keeping all of arxiv.sty's other settings intact and
% just nudging textheight up — gives a few more lines of body per page.
\AtBeginDocument{%
  \newgeometry{
    textheight=9.4in,
    textwidth=6.5in,

  }%

% Compress vertical whitespace for the SPECS 3-page limit.
% Use the classic indented-first-line paragraph style: \parindent makes
% paragraph breaks visible without needing the extra \parskip space that
% would otherwise have to do that work. Keeps the page tight while
% paragraph structure remains legible inside dense blocks of text.

% Tighten the arxiv.sty title block. Three places eat vertical space:
% the rules around the title, and two large \vskips at the end of
% \@maketitle (one between author and date, one between date and the
% abstract). We compress all three to claw back ~0.5 in above the abstract.
\makeatletter
\renewcommand{\@toptitlebar}{\hrule height 1\p@\vskip 0.10in\vskip -\parskip}
\renewcommand{\@bottomtitlebar}{\vskip 0.12in\vskip -\parskip\hrule height 1\p@\vskip 0.05in}
\renewcommand{\@maketitle}{%
  \vbox{%
    \hsize\textwidth
    \linewidth\hsize
    \vskip 0.05in
    \@toptitlebar
    \centering
    {\LARGE\sc \@title\par}
    \@bottomtitlebar
    \textsc{\undertitle}\\
    \vskip 0.05in
    \def\And{%
      \end{tabular}\hfil\linebreak[0]\hfil
      \begin{tabular}[t]{c}\bf\rule{\z@}{20\p@}\ignorespaces
    }%
    \def\AND{%
      \end{tabular}\hfil\linebreak[4]\hfil
      \begin{tabular}[t]{c}\bf\rule{\z@}{20\p@}\ignorespaces
    }%
    \begin{tabular}[t]{c}\bf\rule{\z@}{20\p@}\@author\end{tabular}%
    \vskip 0.10in \@minus 0.05in
    \center{\@date}
  }%
}
% Render the abstract environment one point smaller than body text so it
% sits visually as an abstract and saves vertical space on the title page.
% Drop the visible "Abstract" heading but keep its vertical footprint
% (the abstract is recognisable from its quote-style indentation and
% smaller font, so the heading is redundant).
\renewenvironment{abstract}
  {\centerline{\vphantom{\large \bfseries \scshape Abstract}}\begin{quote}\small}
  {\end{quote}}
