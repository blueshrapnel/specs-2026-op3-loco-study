%% Shared paper metadata.
%% Edit this file to change titles, running page headers, and authors
%% across builds. Two author-block macros are provided because the
%% three main files use two different conventions:
%%   - main.tex and main-extended.tex use the arxiv.sty style:
%%       \author{Name\\Affiliation\\\texttt{email}\And ...}
%%     → \paperAuthorsArxiv
%%   - main-production.tex uses authblk:
%%       \author[1]{Name} ... \affil[1]{Affiliation}
%%     → \paperAuthorsAuthblk plus \paperAffiliationsAuthblk

%% Affiliation shared by all authors — single-affiliation, single-uni venue.
%% Carried as a title-level footnote rather than under each author so the
%% authors fit on one line and we save vertical space.
\newcommand{\paperAffiliation}{University of Hertfordshire}
\newcommand{\paperTitleFootnote}{All authors are with the \paperAffiliation.}
\newcommand{\paperCorrespondingFootnote}{Corresponding author: \texttt{k.archer2@herts.ac.uk}.}
